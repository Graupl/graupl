<script setup>
  import { onBeforeUnmount, useTemplateRef } from "vue";
  import LiveExample from "../vue-components/LiveExample.vue";
  import Tooltip from "../../packages/core/src/js/tooltip/Tooltip.js";

  const preview = useTemplateRef("tooltip-preview");
  let tooltip;

  const exampleTooltip = `
<div class="tooltip">
  <span class="tooltip-description hide">The north gate closes at sundown.</span>
  <button type="button" class="tooltip-toggle" aria-label="North gate hours"></button>
</div>
  `;

  function initializeTooltip() {
    const tooltipElement = preview.value?.querySelector(".example .tooltip");

    if (tooltip?.dom.tooltip === tooltipElement) return;

    tooltip?.dispose();
    tooltip = tooltipElement && new Tooltip({
      tooltipElement,
      tooltipToggleElement: tooltipElement.querySelector(".tooltip-toggle"),
      tooltipDescriptionElement: tooltipElement.querySelector(".tooltip-description"),
      initialize: true,
    });
  }

  onBeforeUnmount(() => tooltip?.dispose());
</script>

# Tooltips

The tooltip component shows a short description alongside a button. Use it for supplemental information and keep essential instructions visible on the page.

## Base classes

Wrap the description and its button in `.tooltip`. Use `.tooltip-description` for the text and `.tooltip-toggle` on a button with an accessible name. Initialization gives the description the tooltip role and associates it with the button through `aria-describedby`.

By default, the tooltip opens when you hover over the button or Tab to it. Click the button or press Enter or Space to toggle it. Press Escape to close it.

::: info Note
The preview limits the description width so it fits beside the button on narrow screens.
:::

<div ref="tooltip-preview">
  <live-example :source-code="exampleTooltip" example-class="tooltip-preview" @updated="initializeTooltip" />
</div>

## Initialization

Run the tooltip generator after the markup is in the DOM.

```js
import generateTooltips from "@graupl/core/tooltip/generator";

generateTooltips();
```
