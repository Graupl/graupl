<script setup>
  import { onBeforeUnmount, useTemplateRef } from "vue";
  import LiveExample from "../vue-components/LiveExample.vue";
  import Accordion from "../../packages/core/src/js/accordion/Accordion.js";

  const basicPreview = useTemplateRef("basic-preview");
  const controlsPreview = useTemplateRef("controls-preview");
  const instances = new Map();

  const items = [
    { title: "North gate", body: "The gate closes at sundown." },
    { title: "East gate", body: "Use this gate to return after sundown." },
  ];

  function renderAccordion(controls = "") {
    const panels = items.map(({ title, body }, index) => {
      const expanded = index === 0;

      return `  <div class="accordion-item ${expanded ? "show" : "hide"}">
    <div class="accordion-item-header">
      <h3 class="accordion-item-title">
        <button type="button" class="accordion-item-toggle" aria-expanded="${expanded}">${title}</button>
      </h3>
    </div>
    <div class="accordion-item-content">
      <div class="accordion-item-body">
        <p>${body}</p>
      </div>
    </div>
  </div>`;
    }).join("\n");

    return `<div class="accordion">
${controls}${panels}
</div>`;
  }

  const basicAccordion = renderAccordion();
  const accordionWithControls = renderAccordion(`  <div class="accordion-control-container">
    <button type="button" class="accordion-control expand-all">Expand all</button>
    <button type="button" class="accordion-control collapse-all">Collapse all</button>
  </div>
`);

  function initializePreview(preview) {
    const accordionElement = preview?.querySelector(".example .accordion");
    const previous = instances.get(preview);

    if (previous?.dom.accordion === accordionElement) return;

    previous?.dispose();
    instances.delete(preview);

    if (accordionElement) {
      instances.set(preview, new Accordion({ accordionElement, initialize: true }));
    }
  }

  onBeforeUnmount(() => {
    instances.forEach((accordion) => accordion.dispose());
  });
</script>

# Accordions

The accordion component provides a set of classes for grouping content into panels that readers can expand or collapse.

## Base classes

Wrap the items in `.accordion`. Each `.accordion-item` contains an `.accordion-item-header` and an `.accordion-item-content` panel. Inside the header, use a heading with `.accordion-item-title` and a button with `.accordion-item-toggle`. Put the main content in `.accordion-item-body`, with an optional `.accordion-item-footer` for extra details.

To start with an item expanded, add `.show` to the item and set `aria-expanded="true"` on its button. Use `.hide` and `aria-expanded="false"` for a collapsed item. Once initialized, click a button or press Enter or Space while it has focus to toggle its panel.

## Initialization

Run the accordion generator after the markup is in the DOM. It links each toggle to its panel and keeps the expanded state in sync as readers use it.

```js
import generateAccordions from "@graupl/core/accordion/generator";

generateAccordions();
```

## Expand and collapse all

Place the buttons inside `.accordion-control-container`. Give each button the `.accordion-control` class and add `.expand-all` or `.collapse-all` to choose its action.

## Options

By default, readers can expand multiple panels and collapse all of them. Set `allowExpandMultiple` to `false` to allow only one expanded panel, or set `allowCollapseAll` to `false` to keep one panel expanded. Start with at least one expanded panel when `allowCollapseAll` is `false`. If both options are `false`, start with exactly one.

Set `optionalKeySupport` to `true` to let readers move between item toggles with the arrow keys, Home, and End.

Pass the options to the generator or set them for a single accordion with `data-graupl-accordion-options`.

```js
generateAccordions({
  options: {
    allowExpandMultiple: false,
    allowCollapseAll: false,
    optionalKeySupport: true,
  },
});
```

## Variants

By default, no theme variants are generated.

If you are [compiling your own version of Graupl](../compiling-graupl.md), you can set the accordion module’s `$themeable` variable to `true` to generate the `.primary`, `.secondary`, and `.tertiary` variants.
